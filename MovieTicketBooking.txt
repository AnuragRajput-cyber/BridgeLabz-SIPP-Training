CLASS Ticket
    ATTRIBUTES:
        movieName
        seatNumber
        price
        booked (true/false)
    METHODS:
        cancel()
            booked = false
            PRINT "Ticket cancelled for seat " + seatNumber

CLASS Theatre
    ATTRIBUTES:
        theatreName
        availableSeats (list)
        ticketsIssued (map of seat → Ticket)
    METHODS:
        bookTicket(movie, seat, price)
            IF seat in availableSeats
                CREATE new Ticket(movie, seat, price)
                REMOVE seat from availableSeats
                ADD ticket to ticketsIssued
                PRINT "Ticket booked"
                RETURN ticket
            ELSE
                PRINT "Seat not available"
                RETURN null

        cancelTicket(seat)
            IF seat exists in ticketsIssued
                CALL ticket.cancel()
                ADD seat back to availableSeats
                REMOVE from ticketsIssued
            ELSE
                PRINT "No booking found"

CLASS Customer
    ATTRIBUTES:
        name
    METHODS:
        bookMovie(theatre, movie, seat)
            CALL theatre.bookTicket(movie, seat, 250)
            PRINT name + " booked seat " + seat

        cancelBooking(theatre, seat)
            CALL theatre.cancelTicket(seat)
            PRINT name + " cancelled seat " + seat

MAIN PROGRAM
    availableSeats = [A10, A11, A12, A13]
    CREATE Theatre("PVR", availableSeats)
    CREATE Customer("Ravi")

    CALL Ravi.bookMovie(PVR, "Avengers: Endgame", "A12")
    CALL Ravi.cancelBooking(PVR, "A12")
